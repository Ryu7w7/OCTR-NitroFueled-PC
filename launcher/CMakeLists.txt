cmake_minimum_required(VERSION 3.20)
project(sacredfire_launcher LANGUAGES CXX)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../out/deps/prefix")
set(IMGUI_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../gpu/third_party/imgui)
if(NOT EXISTS "${IMGUI_DIR}/imgui.cpp")
    message(FATAL_ERROR "Run git submodule update --init gpu/third_party/imgui")
endif()
set(LAUNCHER_SOURCES windows/settings_import.cpp nat_check.cpp
    ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
if(WIN32)
    enable_language(RC)
    list(APPEND LAUNCHER_SOURCES windows/discord_presence.cpp windows/discord_link.cpp windows/port_update.cpp
        ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp
        ../packaging/windows/launcher.rc)
    set(LAUNCHER_LIBRARIES d3d11 dxgi d3dcompiler dwmapi windowscodecs ole32 shell32 uuid winhttp bcrypt advapi32 ws2_32)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2)
    pkg_check_modules(TRAY REQUIRED IMPORTED_TARGET ayatana-appindicator-glib dbusmenu-glib-0.4)
    find_package(PNG REQUIRED)
    find_package(Freetype REQUIRED)
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    list(APPEND LAUNCHER_SOURCES ${IMGUI_DIR}/misc/freetype/imgui_freetype.cpp
        ${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp ${IMGUI_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
    set(LAUNCHER_LIBRARIES PkgConfig::SDL3 PkgConfig::TRAY PNG::PNG Freetype::Freetype OpenSSL::Crypto Threads::Threads)
endif()
add_executable(ctr-launcher launcher.cpp ${LAUNCHER_SOURCES})
target_compile_features(ctr-launcher PRIVATE cxx_std_20)
target_compile_options(ctr-launcher PRIVATE -O2)
target_include_directories(ctr-launcher PRIVATE ${IMGUI_DIR} ${IMGUI_DIR}/backends)
target_link_libraries(ctr-launcher PRIVATE ${LAUNCHER_LIBRARIES})
if(WIN32)
    target_compile_definitions(ctr-launcher PRIVATE UNICODE _UNICODE)
    target_link_options(ctr-launcher PRIVATE -static -municode -s)
    set_target_properties(ctr-launcher PROPERTIES WIN32_EXECUTABLE ON OUTPUT_NAME CTR-NitroFueled-PC
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" INTERPROCEDURAL_OPTIMIZATION OFF)
else()
    target_compile_definitions(ctr-launcher PRIVATE _GNU_SOURCE)
    set_target_properties(ctr-launcher PROPERTIES OUTPUT_NAME sacredfire-launcher)
endif()
# The game build owns its own test targets; standalone CI exercises this target directly.
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    return()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(nat-check-test ../tests/test_nat_check.cpp nat_check.cpp)
    target_compile_features(nat-check-test PRIVATE cxx_std_20)
    target_compile_options(nat-check-test PRIVATE -UNDEBUG)
    target_include_directories(nat-check-test PRIVATE windows)
    if(WIN32)
        target_link_options(nat-check-test PRIVATE -static)
        target_link_libraries(nat-check-test PRIVATE ws2_32 bcrypt)
    else()
        target_link_libraries(nat-check-test PRIVATE OpenSSL::Crypto)
    endif()
    add_test(NAME nat-check COMMAND nat-check-test)
    add_executable(settings-import-test ../tests/test_settings_import.cpp windows/settings_import.cpp)
    target_compile_features(settings-import-test PRIVATE cxx_std_20)
    target_include_directories(settings-import-test PRIVATE windows)
    add_test(NAME settings-import COMMAND settings-import-test)
    if(WIN32)
        target_link_options(settings-import-test PRIVATE -static)
        return()
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME launcher-upgrade COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../packaging/test_launcher_upgrade.py $<TARGET_FILE:ctr-launcher> --native)
    add_executable(launcher-process-test ../tests/test_launcher_process.cpp)
    target_compile_features(launcher-process-test PRIVATE cxx_std_20)
    target_compile_definitions(launcher-process-test PRIVATE _GNU_SOURCE)
    target_compile_options(launcher-process-test PRIVATE -UNDEBUG)
    add_test(NAME launcher-process COMMAND launcher-process-test)
    add_executable(launcher-ui-test ../tests/test_launcher_ui.cpp ${LAUNCHER_SOURCES}
        ../gpu/shim/bbport_settings.cpp ../gpu/shim/bbport_menu_es.cpp ../gpu/shim/bbport_menu_more.cpp)
    target_compile_features(launcher-ui-test PRIVATE cxx_std_20)
    target_compile_definitions(launcher-ui-test PRIVATE _GNU_SOURCE)
    target_compile_options(launcher-ui-test PRIVATE -UNDEBUG -O0)
    target_include_directories(launcher-ui-test PRIVATE ${IMGUI_DIR} ${IMGUI_DIR}/backends)
    target_link_libraries(launcher-ui-test PRIVATE ${LAUNCHER_LIBRARIES})
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME launcher-ui COMMAND ${XVFB_RUN} -a $<TARGET_FILE:launcher-ui-test>
            ${CMAKE_CURRENT_SOURCE_DIR}/.. ${CMAKE_CURRENT_BINARY_DIR}/ui-test)
        set_tests_properties(launcher-ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=x11;LIBGL_ALWAYS_SOFTWARE=1;__GLX_VENDOR_LIBRARY_NAME=mesa")
    endif()
endif()
